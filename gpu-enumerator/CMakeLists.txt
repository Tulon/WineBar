cmake_minimum_required(VERSION 3.13)

project(gpu-enumerator LANGUAGES C)

include(FindPkgConfig)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED TRUE)

# The reason we locate the Vulkan package using pkgconfig rather than
# using find_package() is that Vulkan installs a .pc file but not
# a CMake module configuration file, which makes it easier to have
# a Vulkan installation from the Flutter snap to be located instead
# of the system one. Note that the official Flutter snap doesn't
# install the required libvulkan-dev package, so building with
# the Flutter snap only works with our forked version of that snap.
# We use that forked version in our CI builds. Using the Flutter snap
# (forked or not) makes the binaries generated by Dart / Flutter runnable
# on older Linux distros.
pkg_check_modules(Vulkan REQUIRED IMPORTED_TARGET vulkan)

add_subdirectory(src)
